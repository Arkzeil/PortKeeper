cmake_minimum_required(VERSION 3.16)
project(portkeeper VERSION 0.1.0 LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Portkeeper requires Linux")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(PORTKEEPER_ASAN "Enable AddressSanitizer" OFF)
option(PORTKEEPER_UBSAN "Enable UndefinedBehaviorSanitizer" OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
if(PORTKEEPER_ASAN)
  add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address)
endif()
if(PORTKEEPER_UBSAN)
  add_compile_options(-fsanitize=undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=undefined)
endif()
add_library(portkeeper_core STATIC src/parse.c src/discover.c src/process.c)
target_include_directories(portkeeper_core PUBLIC src)
target_compile_definitions(portkeeper_core PUBLIC _GNU_SOURCE)
add_executable(portkeeper src/main.c)
target_link_libraries(portkeeper PRIVATE portkeeper_core)
include(GNUInstallDirs)
install(TARGETS portkeeper RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
include(CTest)
if(BUILD_TESTING)
  add_executable(unit_tests tests/unit.c)
  target_link_libraries(unit_tests PRIVATE portkeeper_core)
  target_compile_definitions(unit_tests PRIVATE
    PK_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/ss")
  add_test(NAME unit COMMAND unit_tests)
  set_tests_properties(unit PROPERTIES TIMEOUT 15)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:portkeeper>)
  set_tests_properties(integration PROPERTIES TIMEOUT 90)
endif()
